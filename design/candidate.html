Usage Examples

// Fully random candidate (industry picked at random)
CandidateFaker faker = new CandidateFaker();
Candidate c = faker.candidate().build();

// Industry-targeted — job history, skills and certifications match
Candidate techCandidate = faker.candidate().buildForIndustry("technology");

// Seeded — fully deterministic across runs
CandidateFaker seeded = new CandidateFaker(new Random(42L));
Candidate same = seeded.candidate().build(); // always identical

// Query available industries
faker.candidate().availableIndustries();               // ["technology","healthcare",…]

// Query certifications for a given industry (useful in tests)
faker.candidate().availableCertificationNames("finance"); // ["CFA Level I","CPA","FRM",…]
faker.candidate().availableSummaryTemplates();         // all 14 summary templates

// Access fields
System.out.println(c.firstName());                    // "Sarah"
System.out.println(c.email());                        // "sarahjohnson47@yopmail.com"
System.out.println(c.mobileNumber());                 // "+1-415-782-3091"
System.out.println(c.professionalSummary());
// "Senior Software Engineer with 3 years of experience in the Technology industry.
//  Skilled in Java, Kubernetes and AWS, ... Holds a Master of Science in Computer Science."
System.out.println(c.address().country());            // "US"
System.out.println(c.address().addressLine2());       // "Apt 204" or ""

// Education history (1–3 entries, chronological)
c.educationHistory().forEach(edu -> {
    System.out.printf("%s | %s | %s %.1f%n",
        edu.degreeLevel(), edu.institutionName(),
        edu.scoreType(), edu.scoreValue());
});
// UNDERGRADUATE | MIT | CGPA 8.7
// POSTGRADUATE  | Stanford University | PERCENTAGE 82.5

// Job history (1–4 entries, oldest first; last entry is the current job)
c.jobHistory().forEach(job -> {
    System.out.printf("%s @ %s — %s  (%s → %s)%n",
        job.designation(), job.companyName(), job.role(),
        job.startDate(), job.isCurrent() ? "present" : job.endDate());
    job.responsibilities().forEach(r -> System.out.println("  • " + r));
});

// Certifications (0–3, industry-specific)
c.certifications().forEach(cert ->
    System.out.printf("%s  (%s)  issued %s%n",
        cert.name(), cert.issuingOrganization(), cert.issuedDate()));
// AWS Certified Solutions Architect – Associate  (Amazon Web Services)  issued 2023-08-15
